100% Automated WhatsApp Server Bot Script

Use this Node.js script to automatically send discipline reminders to your WhatsApp group 1 minute before duty without needing manual intervention.

Node.js Server Script (WhatsApp Web JS / Twilio) 
// Install: npm install whatsapp-web.js qrcode-terminal node-cron
const { Client, LocalAuth } = require('whatsapp-web.js');
const qrcode = require('qrcode-terminal');
const cron = require('node-cron');

const client = new Client({ authStrategy: new LocalAuth() });

client.on('qr', (qr) => qrcode.generate(qr, { small: true }));
client.on('ready', () => console.log('BIET Discipline WhatsApp Bot Active!'));

// Daily Duty Roster Mapping (Updated Duty Times)
const schedule = {
  "Monday": {
    "09:19": { gf: "Ms. Abhinaya", ff: "Ms. Haripriya", sf: "Ms. K. Vani" },
    "10:19": { gf: "Ms. Adiba", ff: "Mr. Vipin", sf: "Dr. Subhasish" },
    "11:29": { gf: "Dr. Kishore", ff: "Mr. Nagarjuna", sf: "Dr. Agnivo" },
    "13:09": { gf: "Ms. Prashanthi", ff: "Ms. Priyadarshini", sf: "Ms. S. Vani" },
    "14:09": { gf: "Mr. J. V. Ramana", ff: "Ms. S. Radhika", sf: "Dr. U. Srinivas" }
  }
};

cron.schedule('* * * * *', async () => {
  const now = new Date();
  const day = ['Sunday', 'Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday'][now.getDay()];
  const time = now.toTimeString().substring(0, 5);

  if (schedule[day] && schedule[day][time]) {
    const duty = schedule[day][time];
    const message = `🚨 *DISCIPLINE DUTY REMINDER* 🚨\n` +
                    `⏰ *Slot starts in 1 Minute!*\n\n` +
                    `📍 *Ground Floor:* ${duty.gf}\n` +
                    `📍 *First Floor:* ${duty.ff}\n` +
                    `📍 *Second Floor:* ${duty.sf}\n\n` +
                    `Please report immediately to maintain student discipline. Thank you!`;

    const groupJid = 'YOUR_WHATSAPP_GROUP_ID@g.us';
    await client.sendMessage(groupJid, message);
    console.log(`Duty alert dispatched for ${day} at ${time}`);
  }
});

client.initialize();
